def google_detect(text):
    """Detect language using Google Translate API"""
    try:
        url = "https://translation.googleapis.com/language/translate/v2/detect"
        params = {'q': text, 'key': GOOGLE_KEY}
        start = time.time()
        response = requests.get(url, params=params)
        end = time.time()
        
        if response.status_code == 200:
            data = response.json()
            detected = data['data']['detections'][0][0]
            return detected['language'], detected['confidence'], end - start, "success"
        else:
            return None, 0, end - start, f"error_{response.status_code}"
    except Exception as e:
        return None, 0, 0, f"exception_{str(e)[:30]}"

def openai_detect(text, model="gpt-4o-mini", api_key=None):
    """Detect language using OpenAI API"""
    if api_key is None:
        api_key = OPENAI_4o_KEY
    
    try:
        url = "https://api.openai.com/v1/chat/completions"
        headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        payload = {
            "model": model,
            "messages": [
                {"role": "system", "content": "You are a language detection system. Return ONLY the language name and confidence as High/Medium/Low. Format: Language: [language], Confidence: [High/Medium/Low]"},
                {"role": "user", "content": f"What language is this title written in? Title: '{text}'"}
            ],
            "temperature": 0.1
        }
        start = time.time()
        response = requests.post(url, headers=headers, json=payload)
        end = time.time()
        
        if response.status_code == 200:
            data = response.json()
            result = data['choices'][0]['message']['content']
            
            # Parse OpenAI response
            language = "unknown"
            confidence_raw = "Medium"
            if "Language:" in result:
                lang_part = result.split("Language:")[1].split(",")[0].strip()
                language = lang_part.lower()
            if "Confidence:" in result:
                conf_part = result.split("Confidence:")[1].strip()
                confidence_raw = conf_part
            
            # Convert to numeric score
            if confidence_raw == "High":
                confidence = 0.95
            elif confidence_raw == "Medium":
                confidence = 0.70
            else:
                confidence = 0.40
            
            return language, confidence, end - start, "success"
        else:
            return None, 0, end - start, f"error_{response.status_code}"
    except Exception as e:
        return None, 0, 0, f"exception_{str(e)[:30]}"

def get_clean_openai_key():
    raw_key = userdata.get('OPEN_AI_KEY')
    if raw_key:
        return raw_key.strip()
    raise ValueError("OPEN_AI_KEY not found in secrets")

def google_translate(text, dest='es'):
    try:
        url = "https://translate.googleapis.com/translate_a/single"
        params = {
            "client": "gtx",
            "sl": "en",
            "tl": dest,
            "dt": "t",
            "q": text
        }
        response = requests.get(url, params=params, timeout=10)
        if response.status_code == 200:
            result = response.json()
            translated = ''
            for sentence in result[0]:
                if sentence[0]:
                    translated += sentence[0]
            return translated.strip()
        return f"Error: HTTP {response.status_code}"
    except Exception as e:
        return f"Error: {str(e)}"

OPENAI_KEY = get_clean_openai_key()
client = openai.OpenAI(api_key=OPENAI_KEY)

class TranslationComparer:
    def __init__(self, client):
        self.client = client

    def translate_google(self, text: str, dest: str = 'es') -> str:
        return google_translate(text, dest=dest)

    def translate_openai(self, text: str, target_language: str = 'Spanish') -> str:
        try:
            response = self.client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[
                    {"role": "system", "content": f"Translate the following English text to {target_language}. Return only the translation."},
                    {"role": "user", "content": text}
                ],
                temperature=0.3,
                max_tokens=1000
            )
            return response.choices[0].message.content.strip()
        except Exception as e:
            return f"OpenAI Error: {str(e)}"

comparer = TranslationComparer(client)

def run_diclets_test(comparer, test_cases):
    all_results = []
    total_tests = sum(len(texts) for texts in test_cases.values())
    current_test = 0

    print(f"\nStarting test ({total_tests} translations)...")

    for category, texts in test_cases.items():
        for text in texts:
            current_test += 1

            g_start = time.time()
            google_result = comparer.translate_google(text)
            g_time = time.time() - g_start

            time.sleep(0.5)

            o_start = time.time()
            openai_result = comparer.translate_openai(text)
            o_time = time.time() - o_start

            winner = "Google" if g_time < o_time else "OpenAI"

            result = {
                'category': category,
                'word_count': len(text.split()),
                'original_text': text,
                'google_translation': google_result,
                'google_time': round(g_time, 3),
                'openai_translation': openai_result,
                'openai_time': round(o_time, 3),
                'winner': winner
            }
            all_results.append(result)

            time.sleep(0.5)

    print(f"\nCompleted {total_tests} translations")
    return pd.DataFrame(all_results)
